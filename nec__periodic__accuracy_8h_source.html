2// The accuracy policy of the periodic (infinite-array) solver. One fixed
3// tolerance governs it: every scan impedance carries an estimated error, and
4// a configuration whose estimate exceeds the tolerance is refused with a
5// reason, never loosened. Nothing here is exposed as a parameter.
6//
7// The estimated error covers the solver's numerics, relative to the largest
8// scan-impedance entry: the Ewald kernel, the Gauss rule over each source
9// segment and the rounding near a Wood anomaly. Thin-wire modelling error is
10// not included; in particular, NEC's thin-wire kernel (used for close images)
11// and the filament of the smooth kernel (used for the other images) differ by
12// about k a^2 / (2 d), which reaches 1e-5 to 3e-4 for tightly packed or thick
13// wires. See docs/periodic-floquet-evaluation.md, issue #25.
14#include <algorithm>
15#include <cmath>
16
17namespace nec_periodic {
18// Largest estimated error of a result.
19constexpr double tolerance = 1e-5;
20
21// The Ewald kernel, as measured by the kernel tests against a direct lattice
22// sum and the spectral series.
23constexpr double kernel_error = 1e-10;
24
25// The default Gauss rule over a source segment of the given length in
26// wavelengths. Measured on lattices of 0.5, 1 and 3 wavelengths: at most
27// 2e-11 up to a tenth of a wavelength, then growing like length^10 (fastest
28// in small cells); this bound is at least four times the measurement.
29inline double gauss_error(double length) {
30 return 1e-10 * std::max(1.0, std::pow(length / 0.1, 10));
31}
32
33// Work bound of the spectral sum: the Floquet modes of one kernel evaluation.
34// The kernel keeps its accuracy by widening the spectral sum as the cell
35// grows (about 31 modes per square wavelength above a few wavelengths), so a
36// larger cell cannot reach the tolerance within this bound.
37constexpr double max_spectral_modes = 10000;
38}
