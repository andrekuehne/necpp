76inline void add_wires(nec_context& model, const std::vector<nec_wire_definition>& wires)
77{
78 for (const nec_wire_definition& item : wires) {
79 model.get_geometry()->wire(
80 item.tag, item.segments,
81 item.x1, item.y1, item.z1,
82 item.x2, item.y2, item.z2,
83 item.radius_m, 1.0, 1.0);

85}

87inline void build_stateful(
88 nec_stateful_model& model,
89 const std::vector<nec_wire_definition>& wires,
90 const std::vector<nec_port_definition>& ports,
91 nec_ground_connection connection = nec_ground_connection::none,
92 nec_ground_kind ground = nec_ground_kind::free_space,
93 nec_float frequency_mhz = kFrequencyMHz)
94{
95 add_wires(model, wires);
96 model.complete_geometry(connection);
97 model.define_ports(ports);
98 if (ground != nec_ground_kind::free_space)
99 model.set_ground({ ground, 0.0, 0.0 });
100 model.prepare(frequency_mhz);
101}

119inline int segment_in_tag(const c_geometry& geometry, int native_index)
120{
121 const int tag = tag_of(geometry, native_index);
122 int count = 0;
123 for (int64_t index = 0; index <= native_index; ++index) {
124 if (geometry.segment_tags[index] == tag)
125 ++count;
126 }
