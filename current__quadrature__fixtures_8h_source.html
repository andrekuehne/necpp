22// True when NECPP_WRITE_BENCH_EVIDENCE=1 asks the baseline tests to rewrite
23// the committed benchmark evidence.
24inline bool write_committed_bench_evidence()
25{
26#if defined(_MSC_VER)
27 char* value = nullptr;
28 size_t length = 0;
29 const bool requested =
30 _dupenv_s(&value, &length, "NECPP_WRITE_BENCH_EVIDENCE") == 0 &&
31 value != nullptr && std::string(value) == "1";
32 std::free(value);
33 return requested;
34#else
35 const char* value = std::getenv("NECPP_WRITE_BENCH_EVIDENCE");
36 return value != nullptr && std::string(value) == "1";
37#endif
38}

40// Where a native baseline JSON is recorded. The committed evidence under
41// packages/necpp-wasm/bench/evidence/<work_package>/ is rewritten only when
42// NECPP_WRITE_BENCH_EVIDENCE=1; otherwise the file goes to the temp
43// directory, so ordinary test runs leave the source tree unchanged while
44// still exercising the recording.
45inline std::filesystem::path native_baseline_destination(const char* work_package)
46{
47 const std::filesystem::path directory = write_committed_bench_evidence()
48 ? std::filesystem::path(__FILE__).parent_path().parent_path()
49 / "packages" / "necpp-wasm" / "bench" / "evidence"
50 : std::filesystem::temp_directory_path() / "necpp-bench-evidence";
51 return directory / work_package / "native-baseline.json";
52}

122inline void build_stateful(
123 nec_stateful_model& model,
124 const std::vector<nec_wire_definition>& wires,
125 const std::vector<nec_port_definition>& ports,
126 nec_model_ground_connection connection = nec_model_ground_connection::none,
127 nec_ground_kind ground = nec_ground_kind::free_space,
128 nec_float frequency_mhz = kFrequencyMHz)
129{
130 add_wires(model, wires);
131 model.complete_geometry(connection);
132 model.define_ports(ports);
133 if (ground != nec_ground_kind::free_space)
134 model.set_ground({ ground, 0.0, 0.0 });
135 model.prepare(frequency_mhz);
136}

155{
156 const int tag = tag_of(geometry, native_index);
157 int count = 0;
158 for (int64_t index = 0; index <= native_index; ++index) {
