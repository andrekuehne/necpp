69// Cauchy bound on the Jacobi-Anger tail; see docs/ring-far-field-evaluation.md.
70inline double log_tail(double a, int n) {
71 if (a == 0) return -INFINITY;
72 if (n + 1 <= a) return INFINITY;
73 double t = std::acosh((n + 1) / a);
74 return std::log(2.) + a * std::sinh(t) - (n + 1) * t
75 - std::log(-std::expm1(-t));
76}
77inline int cutoff(double a, double eps, int limit) {
78 if (!std::isfinite(a) || !(eps > 0) || a >= limit) return limit;
79 if (a == 0) return 0;
80 int n = int(std::floor(a));
81 while (n < limit && !(log_tail(a,n) <= std::log(eps))) ++n;
82 return n;
83}
