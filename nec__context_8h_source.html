123 void reset_processing_to_structure_loading() {
124 switch (processing_state) {
125 /* If we havent allocated memory, then leave state there. Otherwise set to
126 * structure loading
127 * */
128 case PROCESSING_MEMORY_ALLOC:
129 processing_state = PROCESSING_MEMORY_ALLOC;
130 break;
131 default:
132 processing_state = PROCESSING_STRUCTURE_LOADING;
133 }
134 }
135
136 void reset_processing_to_excitation_setup() {
137 switch (processing_state) {
138 /* If we havent allocated memory, then leave state there. Otherwise set to
139 * structure loading
140 * */
141 case PROCESSING_MEMORY_ALLOC:
142 processing_state = PROCESSING_MEMORY_ALLOC;
143 break;
144 case PROCESSING_STRUCTURE_LOADING:
145 processing_state = PROCESSING_STRUCTURE_LOADING;
146 break;
147 default:
148 processing_state = PROCESSING_EXCITATION_SETUP;
149 }

274 double get_impedance_real(int freq_index = 0, int feed_index = 0) {
275 nec_antenna_input* ipt = get_input_parameters(freq_index);
276 if (NULL == ipt) return -999.0;
277 vector<nec_complex>& imp(ipt->get_impedance());
278 if (feed_index < 0 || (size_t)feed_index >= imp.size()) return -999.0;
279 return imp[feed_index].real();
280 }
286 double get_impedance_imag(int freq_index = 0, int feed_index = 0) {
287 nec_antenna_input* ipt = get_input_parameters(freq_index);
288 if (NULL == ipt) return -999.0;
289 vector<nec_complex>& imp(ipt->get_impedance());
290 if (feed_index < 0 || (size_t)feed_index >= imp.size()) return -999.0;
291 return imp[feed_index].imag();
