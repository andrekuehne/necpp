1076 // The radiation vector N = int J(r') exp(j (kt, kz) . r') dr' of the
1077 // unit-cell current of the latest solve, with its ground image over
1078 // perfect ground, in wavelength units and native current coefficients
1079 // (amperes per wavelength).
1080 // With reference_z h, the direct current is weighed against
1081 // exp(j (kt . rho + kz (z - h))) and its image against
1082 // exp(j (kt . rho - kz (z + h))), so N comes out times exp(-j kz h).
1083 // Without ground_image, the direct current alone.
1084 std::array<nec_complex, 3> periodic_radiation_vector(
1085 nec_periodic::vec2 kt, nec_complex kz, nec_float reference_z = 0, bool ground_image = true) const;
1086 // The singular parts of the Floquet modes split out of m_periodic_green
1087 // (near cut-off), and the layered stack's terms that are large (near
1088 // cut-off or a guided-wave pole), solved by a low-rank update; null when
1089 // there are none.
1090 std::shared_ptr<nec_periodic::low_rank_update> m_periodic_low_rank;
1091 // Lattice translations, in wavelengths, under which some periodic image
1092 // comes close to the unit cell. For each observation point and source
1093 // segment, the images within reach go through efld_compute instead of the
1094 // smooth kernel, like the ground image.
1095 struct periodic_close_image {
1096 nec_float x = 0, y = 0;
1097 };
1098 std::vector<periodic_close_image> m_periodic_close_images;

1100 // The periodic fill is split into parts so that a scan change at a fixed
1101 // frequency keeps the scan-independent ones (periodic_fill): the free-space
1102 // fill, and the contribution of each lattice point rho to the sum over
1103 // images, which changes with the scan state only through the phase
1104 // exp(-j kt0 . rho). cmset/cmww fill the part m_fill_part.
1105 enum class fill_part { free_space, periodic_point, periodic_spectral };
1106 fill_part m_fill_part = fill_part::free_space;
1107 // A lattice point in wavelengths; close when some periodic image under it
1108 // comes close to the unit cell (m_periodic_close_images).
1109 struct periodic_fill_point {
1110 nec_float x = 0, y = 0;
1111 bool close = false;
1112 };
1113 // The lattice point of fill_part::periodic_point.
1114 const periodic_fill_point* m_fill_point = nullptr;
1115 struct periodic_scan_independent_fill {
1116 nec_float frequency_mhz = 0;
1117 complex_array free_space;
1118 std::vector<periodic_fill_point> points;
1119 // One matrix per image, or empty when they would take more than
1120 // periodic_fill_cache_bytes; then each scan state fills them again.
1121 std::vector<complex_array> point_matrices;
1122 // The spatial terms from tables in the distance (null on the reference
1123 // fill), and the spectral sum's geometry: its kernel samples and their
