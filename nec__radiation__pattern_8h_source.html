268private:
269 nec_float mean(const real_matrix& pattern) const {
270 nec_float sum = 0.0;
271 int64_t rows = pattern.rows();
272 int64_t cols = pattern.cols();
273 for (int64_t i=0;i<rows;i++) {
274 for (int64_t j=0;j<cols;j++) {
275 sum += pattern(i,j) * _averaging_scales(i,j);
276 }
277 }
278 int64_t len = rows*cols;
279 return sum/(nec_float(len*2) / pi());
280 }
281
282 nec_float sd(const real_matrix& pattern, nec_float _mean) const {
283 nec_float sum = 0.0;
284 int64_t rows = pattern.rows();
285 int64_t cols = pattern.cols();
286
287 for (int64_t i=0;i<rows;i++) {
288 for (int64_t j=0;j<cols;j++) {
289 nec_float diff = pattern(i,j) - _mean;
290 sum += diff*diff * _averaging_scales(i,j);
291 }
292 }
293 int64_t len = rows*cols;
294 return std::sqrt(sum/(nec_float(len*2) / pi()));
295 }

406 nec_float get_power_gain_rhcp(int theta_index, int phi_index) const {
407 nec_float a = get_pol_axial_ratio(theta_index, phi_index);
408 nec_float dbi = get_power_gain_tot(theta_index, phi_index);
409 if (get_pol_sense(theta_index, phi_index) == POL_RIGHT)
410 a = -a;
411
412 nec_float f = (1-2*a+a*a)/(2*(1+a*a));
413 return dbi + 10*log10(f);
414 }
415
418 nec_float get_power_gain_lhcp(int theta_index, int phi_index) const {
419 nec_float a = get_pol_axial_ratio(theta_index, phi_index);
420 nec_float dbi = get_power_gain_tot(theta_index, phi_index);
421 if (get_pol_sense(theta_index, phi_index) == POL_RIGHT)
422 a = -a;
423
424 nec_float f = (1+2*a+a*a)/(2*(1+a*a));
425 return dbi + 10*log10(f);
426 }
