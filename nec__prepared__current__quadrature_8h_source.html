113inline size_t nec_prepared_quadrature_identity_bytes(size_t n_segments)
114{
115 return 12u * n_segments;
116}
117
118inline size_t nec_prepared_quadrature_identity_pad_bytes(size_t n_segments)
119{
120 const size_t unpadded =
121 nec_prepared_quadrature_header_bytes +
122 nec_prepared_quadrature_identity_bytes(n_segments);
123 return (8u - (unpadded % 8u)) % 8u;
124}

126inline size_t nec_prepared_quadrature_sample_count(
127 size_t n_segments, size_t n_nodes, size_t n_image_planes)
128{
129 return n_segments * n_nodes * n_image_planes;
130}
131
132inline size_t nec_prepared_quadrature_geometry_bytes(
133 size_t n_segments, size_t n_nodes, size_t n_image_planes)
134{
135 return 9u * nec_prepared_quadrature_sample_count(
136 n_segments, n_nodes, n_image_planes) * 8u;
137}
138
139inline size_t nec_prepared_quadrature_current_bytes(
140 size_t n_modes, size_t n_segments, size_t n_nodes, size_t n_image_planes)
141{
142 return 2u * n_modes * nec_prepared_quadrature_sample_count(
143 n_segments, n_nodes, n_image_planes) * 8u;
144}
145
146inline size_t nec_prepared_quadrature_packed_bytes(
147 size_t n_modes, size_t n_segments, size_t n_nodes, size_t n_image_planes)
148{
149 return nec_prepared_quadrature_header_bytes +
150 nec_prepared_quadrature_identity_bytes(n_segments) +
151 nec_prepared_quadrature_identity_pad_bytes(n_segments) +
152 nec_prepared_quadrature_geometry_bytes(
153 n_segments, n_nodes, n_image_planes) +
154 nec_prepared_quadrature_current_bytes(
155 n_modes, n_segments, n_nodes, n_image_planes);
156}
