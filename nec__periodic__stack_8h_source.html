2// A layered stack above or below the wires of a periodic unit cell (a radome,
3// a substrate, finite ground), as one Floquet mode sees it: layers uniform in
4// the lattice plane act on each mode only through a transmission line per
5// polarization, TE or TM with respect to z (docs/periodic-floquet-evaluation.md,
6// ADR 0001).
7//
8// The line's voltage is the tangential electric field. Its characteristic
9// admittance is kz / k for TE and impedance kz / (eps k) for TM, so neither
10// becomes infinite when a mode in air reaches cut-off (kz = 0). Time
11// dependence exp(+jwt): a lossy layer has eps = eps_r (1 - j tan_delta), and
12// every kz takes the branch Im(kz) <= 0.
13#include <cmath>
14#include <complex>
15#include <limits>
16#include <vector>

19using complex_d = std::complex<double>;

21struct dielectric_layer {
22 // In the length unit of 1 / k.
23 double thickness = 0;
24 // Relative permittivity (mu_r = 1).
25 complex_d permittivity = 1.0;
26};

28enum class polarization { te, tm };

30// With a stack present, a mode exactly at cut-off (kz = 0) is evaluated at the
31// nearest representable state, |kz| = sqrt(eps) k: there the stack's terms
32// nearly cancel the kernel's singular part, which is infinite at kz = 0.
33inline complex_d stack_cutoff_kz(complex_d kz, double k) {
34 return kz == complex_d(0.0, 0.0)
35 ? complex_d(std::sqrt(std::numeric_limits<double>::epsilon()) * k, 0.0) : kz;
36}

38// What lies beyond the stack's far face: a half-space of the given
39// permittivity (air by default; finite ground), or a perfect conductor.
40struct stack_termination {
41 complex_d permittivity = 1.0;
42 bool perfect_conductor = false;
43};
44
45// One mode incident from the air on the stack's near face. A stack below the
46// wires is the mirror image of one above, so both list their layers from the
47// near face outward.
48struct stack_response {
49 // The reflected over the incident voltage, both at the near face.
50 complex_d reflection;
51 // The outgoing voltage at the far face over the incident one at the near
52 // face, with air beyond the stack; equal to the field amplitude ratio,
53 // since that air has the same kz as the air in front. Zero for any other
54 // termination.
55 complex_d transmission;
56 // Power that does not come back, less what is transmitted into air: the
57 // power left in the layers, and in a half-space beyond them all that
58 // enters it. As a fraction of |a|^2 / 2 with a the incident wave's
59 // power-normalized amplitude at the near face: for a propagating mode the
60 // absorbed fraction of the incident power. Zero exactly at cut-off.
61 double absorbed = 0;
62};
63
64// Layers from the near face outward. k is the wavenumber in air and kz the
65// mode's sqrt(k^2 - |kt|^2) in air (Im(kz) <= 0), the caller's value, so that
66// an exact cut-off stays exact.
67stack_response stack_response_of(const std::vector<dielectric_layer>& layers,
68 double k, complex_d kz, polarization mode, const stack_termination& beyond = {});
