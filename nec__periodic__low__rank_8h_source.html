2// The singular parts of the Floquet modes near cut-off, split out of the
3// periodic kernel (periodic_green::split_modes), enter the interaction matrix
4// as a low-rank update
5//
6// Z = Zr + sum_l c_l u_l v_l^T,
7//
8// u over the observation (row) index, v over the basis (column) index, with
9// coefficients c_l of size 1 / kz. Only Zr is factored; the Woodbury identity
10//
11// Z^-1 b = x0 - W (C^-1 + V^T W)^-1 V^T x0, x0 = Zr^-1 b, W = Zr^-1 U,
12//
13// involves only C^-1, which is finite and zero exactly on cut-off, so port
14// quantities stay finite and continuous through a Wood anomaly.
15//
16// Several modes at cut-off at once (broadside on a square lattice of one
17// wavelength) can couple to the unit cell through linearly dependent columns
18// u, which makes the capacitance matrix C^-1 + V^T W singular exactly there.
19// Its solve is a pseudo-inverse that drops singular values below
20// rank_threshold of the largest; the dropped directions are those that U maps
21// to zero.
22#include <complex>
23#include <functional>
24#include <vector>
25
26namespace nec_periodic {
27class low_rank_update {
28public:
29 using vector = std::vector<std::complex<double>>;
30 static constexpr double rank_threshold = 1e-11;
31
32 // Adds the column pair (u, v) with coefficient c = 1 / inverse_coefficient
33 // (inverse_coefficient = 0 exactly on cut-off), scaled to unit columns; a
34 // zero column adds nothing. All columns have the matrix size.
35 void add(vector u, vector v, std::complex<double> inverse_coefficient);
36 bool empty() const { return columns_.empty(); }
37 // Once Zr is factored: solve_regular(b) overwrites b with Zr^-1 b.
38 void factor(const std::function<void(vector&)>& solve_regular);
39 // Turns x = Zr^-1 b into Z^-1 b in place (the first matrix-size entries).
40 void correct(std::complex<double>* x) const;
41
42private:
43 struct column {
44 vector u, v, w; // w = Zr^-1 u
45 std::complex<double> inverse_coefficient;
46 };
47 std::vector<column> columns_;
48 // Pseudo-inverse of the capacitance matrix C^-1 + V^T W, row-major.
49 vector capacitance_inverse_;
50};
51}
