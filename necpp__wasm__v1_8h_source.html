277/*
278 * Periodic (infinite-array) mode, additive ABI v1 symbols. The completed
279 * geometry becomes the unit cell of an infinite array with lattice vectors
280 * a1, a2 in metres in the xy-plane; port impedances become scan impedances.
281 * Scan phases are the radians by which the current advances from one cell to
282 * the next along a1 and a2; setting a lattice resets them to zero. Both calls
283 * invalidate the factorization. Symmetric geometry and finite ground are
284 * refused, as are far-field and embedded-field calls. set_lattice returns
285 * NECPP_WASM_V1_GEOMETRY_ERROR when a conductor touches or nearly touches a
286 * periodic image (connected arrays are not supported) or, over perfect
287 * ground, the ground plane. prepare returns NECPP_WASM_V1_GEOMETRY_ERROR for
288 * a unit cell too large or segments too long for the periodic tolerance.
289 * Scan states on a Wood anomaly are solved and flagged
290 * (NECPP_WASM_V1_POINT_STATUS_NEAR_CUTOFF).
291 */
292int32_t necpp_wasm_v1_set_lattice(
293 necpp_wasm_v1_model* model,
294 double a1x_m, double a1y_m, double a2x_m, double a2y_m);
295int32_t necpp_wasm_v1_clear_lattice(necpp_wasm_v1_model* model);
296int32_t necpp_wasm_v1_set_scan_phases(
297 necpp_wasm_v1_model* model, double psi1_rad, double psi2_rad);
298/*
299 * Layered stack above the wires of a periodic unit cell (a radome), additive
300 * ABI v1 symbol: its lower face at bottom_z_m and layer_count layers from
301 * there up, each three doubles in `layers` (thickness in metres, relative
302 * permittivity of at least 1, loss tangent), with air above them;
303 * layer_count 0 removes the stack. Invalidates the factorization. Returns
304 * NECPP_WASM_V1_GEOMETRY_ERROR when a conductor enters the stack; prepare
305 * returns NECPP_WASM_V1_GEOMETRY_ERROR for a gap too small for the periodic
306 * tolerance, with the gap and mode count in the message.
307 */
308int32_t necpp_wasm_v1_set_layered_stack_above(
309 necpp_wasm_v1_model* model,
310 double bottom_z_m, const double* layers, size_t layer_count);
311/*
312 * Layered stacks above and below the wires of a periodic unit cell, additive
313 * ABI v1 symbol; replaces both. Above as necpp_wasm_v1_set_layered_stack_above.
314 * Below (a substrate): its upper face at below_top_z_m and below_count layers
315 * from there down, each three doubles as above; below them the ground, with
316 * air between the stack and z = 0, or air in free space. A count of 0
317 * leaves that side without a stack. Returns NECPP_WASM_V1_GEOMETRY_ERROR when
318 * a conductor enters a stack or the lower stack reaches below z = 0 over
319 * ground.
320 */
321int32_t necpp_wasm_v1_set_layered_stacks(
322 necpp_wasm_v1_model* model,
323 double above_bottom_z_m, const double* above_layers, size_t above_count,
324 double below_top_z_m, const double* below_layers, size_t below_count);
